const TelegramBot = require('node-telegram-bot-api'); const express = require('express'); const cors = require('cors'); // Токен твоего бота и URL твоего красивого сайта на GitHub Pages const TOKEN = process.env.BOT_TOKEN || 'ТВОЙ_ТОКЕН_БОТА'; const GITHUB_PAGES_URL = 'https://egorsheva77.github.io/design-app/'; const bot = new TelegramBot(TOKEN, { polling: true }); const app = express(); app.use(express.json()); app.use(cors()); // Главный фокус: когда кнопка обращается к Render, мы мгновенно перекидываем её на GitHub Pages app.get('/', (req, res) => { res.redirect(301, GITHUB_PAGES_URL); }); // Логика кнопки /start в чате bot.onText(/\/start/, (msg) => { const chatId = msg.chat.id; bot.sendMessage(chatId, 'Привет! Нажми на кнопку ниже, чтобы открыть каталог и оформить заказ: 👇', { reply_markup: { keyboard: [ [{ text: 'Заказать дизайн 🚀', web_app: { url: GITHUB_PAGES_URL } }] ], resize_keyboard: true } }); }); // Прием данных из формы заказа обратно в бот app.post('/web-data', async (req, res) => { const { queryId, products, totalPrice } = req.body; try { await bot.answerWebAppQuery(queryId, { type: 'article', id: queryId, title: 'Успешная оплата', input_message_content: { message_text: `🎉 Заказ оформлен успешно!\nСумма заказа: ${totalPrice}₽` } }); return res.status(200).json({}); } catch (e) { return res.status(500).json({}); } }); const PORT = process.env.PORT || 8000; app.listen(PORT, () => { console.log('Сервер запущен на порту ' + PORT); });
